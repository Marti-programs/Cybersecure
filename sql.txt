-- Tworzenie tabeli użytkowników
CREATE TABLE Users (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    Username NVARCHAR(50) NOT NULL UNIQUE,
    PasswordHash NVARCHAR(256) NOT NULL,
    Salt NVARCHAR(128) NOT NULL,
    Role NVARCHAR(50) NOT NULL DEFAULT 'User', -- np. 'Administrator', 'User'
    IsBlocked BIT NOT NULL DEFAULT 0,
    IsFirstLogin BIT NOT NULL DEFAULT 1,
    PasswordChangedDate DATETIME2 NOT NULL DEFAULT GETDATE(),
    TempPasswordExpiry DATETIME2 NULL
);

-- Tabela historii haseł (zapobiega ponownemu użyciu ostatnich haseł)
CREATE TABLE PasswordHistory (
    Id INT IDENTITY(1,1) PRIMARY KEY,
    UserId INT FOREIGN KEY REFERENCES Users(Id) ON DELETE CASCADE,
    PasswordHash NVARCHAR(256) NOT NULL,
    Salt NVARCHAR(128) NOT NULL,
    CreatedAt DATETIME2 NOT NULL DEFAULT GETDATE()
);

-- Tabela ustawień systemowych (polityka haseł, ważność, czas bezczynności)
CREATE TABLE SystemSettings (
    Id INT PRIMARY KEY CHECK (Id = 1), -- Tabela ma przechowywać tylko jeden wiersz konfiguracyjny
    PolicyEnabled BIT NOT NULL DEFAULT 1,
    ExpiryDays INT NOT NULL DEFAULT 30,
    SessionTimeoutMinutes INT NOT NULL DEFAULT 15
);
